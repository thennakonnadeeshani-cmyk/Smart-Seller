function fmt(l, overrideCur){
  const c = overrideCur || curPref;
  let amt = l.cur==='LKR' ? l.price : l.price*RATE;
  if(c==='USD') amt = l.cur==='USD' ? l.price : l.price/RATE;
  const n = new Intl.NumberFormat(undefined,{maximumFractionDigits:0}).format(Math.round(amt));
  return (c==='LKR'?'LKR ':'$')+n + (c!==l.cur?` <small style="color:var(--muted);font-weight:400">(≈${l.cur} ${new Intl.NumberFormat().format(l.price)})</small>`:'');
}

function showPage(p){
  document.querySelectorAll('main > section').forEach(s=>s.classList.add('hidden'));
  $('page-'+p).classList.remove('hidden');
  document.querySelectorAll('#mainNav a').forEach(a=>a.classList.toggle('active',a.dataset.page===p));
  if(p==='buy')renderListings();
  ['properties','vehicles','business'].forEach(cp=>{
    if(p===cp){const grid=document.querySelector('#page-'+cp+' .catPage');
      grid.innerHTML = listings.filter(l=>l.cat===grid.dataset.cat&&l.status==='live').map(cardHTML).join('')||'<p>No listings yet.</p>'}
  });
  if(p==='admin')renderAdmin();
  window.scrollTo(0,0);
}

function cardHTML(l){
  return `<div class="card listing">
    <div class="img">${l.icon||'📦'}</div>
    <div><span class="badge badge-cat">${l.cat}</span> ${l.verified?'<span class="badge badge-verified">✔ Verified</span>':''}</div>
    <h3 style="margin-top:6px">${l.title}</h3>
    <div class="price">${fmt(l)}</div>
    <div class="loc">📍 ${l.city}, ${l.country} • Seller: ${l.seller}</div>
    <p style="font-size:.83rem;color:var(--muted);margin-bottom:10px">${l.desc}</p>
    <div class="actions">
      <a class="btn btn-sm" href="tel:${l.phone}">📞 Call Seller</a>

function renderListings(){
  const q=($('searchQ').value||'').toLowerCase(), c=$('searchCat').value, co=$('searchCountry').value, s=$('searchSort').value;
  let r=listings.filter(l=>l.status==='live');
  if(q)r=r.filter(l=>(l.title+l.desc+l.city).toLowerCase().includes(q));
  if(c)r=r.filter(l=>l.cat===c);
  if(co)r=r.filter(l=>l.country===co);
  if(s==='low')r=[...r].sort((a,b)=>(a.cur==='LKR'?a.price:a.price*RATE)-(b.cur==='LKR'?b.price:b.price*RATE));
  if(s==='high')r=[...r].sort((a,b)=>(b.cur==='LKR'?b.price:b.price*RATE)-(a.cur==='LKR'?a.price:a.price*RATE));
  $('listingGrid').innerHTML = r.map(cardHTML).join('')||'<p>No listings match your search.</p>';
}
function renderAdmin(){
  $('admListings').textContent=listings.length;
  $('admPending').textContent=listings.filter(l=>l.status==='pending').length;
  let revLKR=0;listings.forEach(l=>revLKR+= (l.cur==='LKR'?l.price:l.price*RATE)*0.05);
  $('admRevenue').textContent='LKR '+new Intl.NumberFormat().format(Math.round(revLKR));
  $('admTable').innerHTML=listings.map(l=>`<tr>
    <td>${l.icon||''} ${l.title}</td><td>${l.seller}</td><td>${l.cur==='LKR'?'LKR':'$'}${new Intl.NumberFormat().format(l.price)}</td>
    <td>${l.cur==='LKR'?'LKR':'$'}${new Intl.NumberFormat().format(Math.round(l.price*0.05))}</td>
    <td>${l.status==='live'?'<span class="badge badge-verified">Live</span>':'<span class="badge badge-cur">Pending</span>'}</td>

}

function sellStep(n){
  if(n>1){
    const req=[['s_name','Full name'],['s_phone','Contact number'],['s_whatsapp','WhatsApp'],['s_email','Email'],['s_country','Country'],['s_city','City']];
    if(n>=2){for(const [id,label] of req){const el=$(id);if(!el.value.trim()){toast('⚠️ Please fill: '+label);sellStep(1);return}}}
    if(n===3){const p=['p_title','p_desc','p_price','p_cat'];for(const id of p){const el=$(id);if(!el||!el.value.trim()){toast('⚠️ Please complete product details first');sellStep(2);return}}}
  }
  [1,2,3].forEach(i=>$('sellStep'+i).classList.toggle('hidden',i!==n));
  document.querySelectorAll('#sellTabs button').forEach((b,i)=>b.classList.toggle('active',i===n-1));
  if(n===3&&$('agreeChk').checked)$('agreeStamp').textContent='Accepted: '+new Date().toLocaleString();
}
function calcFee(){
  const p=parseFloat($('p_price').value), c=$('p_cur').value;
  if(!p||p<=0){$('feeAmount').textContent='—';$('feeUsd').textContent='';return}
  const fee=p*0.05;
  $('feeAmount').textContent=(c==='LKR'?'LKR ':'$')+new Intl.NumberFormat(undefined,{maximumFractionDigits:2}).format(fee);
  $('feeUsd').textContent = c==='LKR' ? `≈ USD ${(fee/RATE).toFixed(2)} (indicative rate)` : `≈ LKR ${new Intl.NumberFormat().format(Math.round(fee*RATE))} (indicative rate)`;
}
